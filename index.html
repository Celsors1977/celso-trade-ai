// ============================================= // SISTEMA COMPLETO DE SINAIS CALL / PUT // Versão Autônoma - Tudo incluído // ============================================= function calculateMACD(closes, fast = 12, slow = 26, signalPeriod = 9) { if (!closes || closes.length < slow + 10) { return { macd: 0, signal: 0, histogram: 0 }; } // Função EMA function EMA(data, period) { let multiplier = 2 / (period + 1); let ema = []; let sum = 0; for (let i = 0; i < period; i++) { sum += data[i]; } let prevEMA = sum / period; ema.push(prevEMA); for (let i = period; i < data.length; i++) { prevEMA = (data[i] - prevEMA) * multiplier + prevEMA; ema.push(prevEMA); } return ema; } const emaFast = EMA(closes, fast); const emaSlow = EMA(closes, slow); const macdLine = []; const start = emaFast.length - emaSlow.length; for (let i = 0; i < emaSlow.length; i++) { macdLine.push(emaFast[i + start] - emaSlow[i]); } const signalLine = EMA(macdLine, signalPeriod); const lastMacd = macdLine[macdLine.length - 1] || 0; const lastSignal = signalLine[signalLine.length - 1] || 0; return { macd: lastMacd, signal: lastSignal, histogram: lastMacd - lastSignal }; } // ====================== FUNÇÃO PRINCIPAL ====================== function getTradeSignal(data) { try { const { closes = [], ema9 = 0, ema21 = 0, rsi = 50, volume = 0, avgVolume = 0, ema200 = null } = data || {}; // Proteções if (closes.length < 50 || ema9 === 0 || ema21 === 0) { return null; } const macdData = calculateMACD(closes); const emaDiff = ema9 - ema21; const isBullish = emaDiff > 0; const isBearish = emaDiff < 0; const macdStrength = Math.abs(macdData.macd); const isTrendStrong = macdStrength > 0.0005; // ==================== CALL ==================== if ( isBullish && rsi > 33 && rsi < 58 && macdData.macd > 0 && isTrendStrong ) { return { direction: "CALL", trend: "ALTA", strength: Math.min(100, Math.round(macdStrength * 9000 + 30)), message: "✅ SINAL CALL" }; } // ==================== PUT ==================== if ( isBearish && rsi > 42 && rsi < 67 && macdData.macd < 0 && isTrendStrong ) { return { direction: "PUT", trend: "BAIXA", strength: Math.min(100, Math.round(macdStrength * 9000 + 30)), message: "✅ SINAL PUT" }; } return null; } catch (error) { console.error("Erro no getTradeSignal:", error); return null; } } // ====================== EXEMPLO DE USO ====================== /* const signal = getTradeSignal({ closes: arrayDePrecos, // ← seu array de closes ema9: suaEma9, ema21: suaEma21, rsi: suaRSI, volume: volumeAtual, avgVolume: mediaVolume, ema200: suaEma200 });
