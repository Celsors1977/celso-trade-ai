/** * getTradeSignal - Versão Segura e Completa */ function getTradeSignal(data) { try { const { closes = [], ema9 = 0, ema21 = 0, rsi = 50, volume = 0, avgVolume = 0, ema200 = null } = data || {}; // Proteção contra dados inválidos if (!closes || closes.length < 30) { return null; } // ====================== CÁLCULOS ====================== const macdData = calculateMACD ? calculateMACD(closes) : { macd: 0, histogram: 0 }; const emaDiff = ema9 - ema21; const isBullishEMA = emaDiff > 0; const isBearishEMA = emaDiff < 0; const macdStrength = Math.abs(macdData.macd || 0); const histStrength = Math.abs(macdData.histogram || 0); const isTrendStrong = macdStrength > 0.0005 && histStrength > 0.0002; const isHistogramConfirming = (isBullishEMA && (macdData.histogram || 0) > 0) || (isBearishEMA && (macdData.histogram || 0) < 0); const hasStrongVolume = avgVolume > 0 ? volume > (avgVolume * 1.12) : true; let isAlignedWithHigherTF = true; if (ema200 !== null && ema200 !== 0) { isAlignedWithHigherTF = (isBullishEMA && ema9 > ema200) || (isBearishEMA && ema9 < ema200); } // ====================== SINAL CALL ====================== if ( isBullishEMA && rsi > 32 && rsi < 58 && isTrendStrong && (macdData.macd || 0) > 0 && isHistogramConfirming && hasStrongVolume && isAlignedWithHigherTF ) { return { direction: "CALL", trend: "ALTA", strength: Math.min(100, Math.round(macdStrength * 10000 + 20)), message: "Sinal CALL detectado" }; } // ====================== SINAL PUT ====================== if ( isBearishEMA && rsi > 42 && rsi < 68 && isTrendStrong && (macdData.macd || 0) < 0 && isHistogramConfirming && hasStrongVolume && isAlignedWithHigherTF ) { return { direction: "PUT", trend: "BAIXA", strength: Math.min(100, Math.round(macdStrength * 10000 + 20)), message: "Sinal PUT detectado" }; } return null; } catch (error) { console.error("Erro no getTradeSignal:", error); return null; } }
